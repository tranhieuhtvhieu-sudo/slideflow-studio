    function enterPinDigit(digit) {
      if (currentPinBuffer.length >= 4) return;
      currentPinBuffer += digit;
      updatePinDots();

      if (currentPinBuffer.length === 4) {
        const savedPin = localStorage.getItem(STORAGE_KEYS.PIN) || "2011";
        if (currentPinBuffer === savedPin) {
          sessionStorage.setItem(STORAGE_KEYS.UNLOCKED, "true");
          document.getElementById("pin-error-msg").textContent = "";
          unlockAppUI();
          showToast("Mở khóa thành công! Chào mừng quay lại SlideFlow Studio.", "success");
        } else {
          document.getElementById("pin-error-msg").textContent = "Mã PIN chưa đúng (Mặc định: 2011)";
          setTimeout(() => {
            currentPinBuffer = "";
            updatePinDots();
          }, 450);
        }
      }
    }

    function getDeadlineCountdownInfo(deadlineIso, status) {
      if (status === "Hoàn thành") {
        return {
          text: "Đã bàn giao",
          badgeClass: "bg-emerald-50 text-emerald-700 border border-emerald-200",
          cardBorder: "border-slate-200/70"
        };
      }
      const diffMs = new Date(deadlineIso).getTime() - Date.now();
      if (isNaN(diffMs)) {
        return {
          text: "Chưa rõ hạn",
          badgeClass: "bg-slate-100 text-slate-600",
          cardBorder: "border-slate-200/70"
        };
      }
      if (diffMs <= 0) {
        const overdueHrs = Math.abs(Math.floor(diffMs / 3600000));
        return {
          text: `Quá hạn ${overdueHrs}h`,
          badgeClass: "bg-rose-600 text-white animate-soft-pulse",
          cardBorder: "border-rose-300"
        };
      }
      const totalMins = Math.floor(diffMs / 60000);
      const hrs = Math.floor(totalMins / 60);
      const mins = totalMins % 60;
      if (hrs < 12) {
        return {
          text: `Còn ${hrs}h ${mins}p`,
          badgeClass: "bg-amber-100 text-amber-800 border border-amber-300",
          cardBorder: "border-amber-300"
        };
      }
      const days = Math.floor(hrs / 24);
      const remHrs = hrs % 24;
      return {
        text: days > 0 ? `Còn ${days}d ${remHrs}h` : `Còn ${hrs}h`,
        badgeClass: "bg-sky-50 text-sky-700 border border-sky-200",
        cardBorder: "border-slate-200/70"
      };
    }

    function handleCustomQrUpload(event) {
      const file = event.target.files && event.target.files[0];
      if (!file) return;

      const reader = new FileReader();
      reader.onload = function (e) {
        const img = new Image();
        img.onload = function () {
          // Compress image via canvas so it fits neatly in localStorage
          const canvas = document.createElement("canvas");
          const maxDim = 600;
          let w = img.width;
          let h = img.height;
          if (w > maxDim || h > maxDim) {
            if (w > h) {
              h = Math.round((h * maxDim) / w);
              w = maxDim;
            } else {
              w = Math.round((w * maxDim) / h);
              h = maxDim;
            }
          }
          canvas.width = w;
          canvas.height = h;
          const ctx = canvas.getContext("2d");
          ctx.drawImage(img, 0, 0, w, h);
          const compressedDataUrl = canvas.toDataURL("image/jpeg", 0.86);

          try {
            localStorage.setItem(STORAGE_KEYS.CUSTOM_QR, compressedDataUrl);
            customQrDataUrl = compressedDataUrl;
            qrDisplayMode = "custom";
            localStorage.setItem(STORAGE_KEYS.QR_MODE, "custom");
            renderPaymentInputs();
            updateQuotePreview();
            showToast("Đã tải lên & lưu ảnh QR cá nhân của shop!", "success");
          } catch (storageErr) {
            showToast("Ảnh quá nặng, vui lòng chọn ảnh dung lượng nhỏ hơn.", "error");
          }
        };
        img.src = e.target.result;
      };
      reader.readAsDataURL(file);
    }

    function syncFromGoogleSheet(silent = false) {
      const scriptUrl = localStorage.getItem(STORAGE_KEYS.SCRIPT_URL);
      if (!scriptUrl) {
        if (!silent) showToast("Bạn chưa dán URL Google Apps Script ở tab Bảng Giá & Kết Nối Form!", "info");
        return;
      }

      setSyncUiState("syncing", "Đang đồng bộ...");
      if (!silent) showToast("Đang kéo dữ liệu mới nhất từ Google Sheets...", "info");

      // JSONP Request (bypasses CORS cleanly on GitHub Pages)
      const callbackName = "sfJsonpCb_" + Date.now();
      const scriptTag = document.createElement("script");
      let isCompleted = false;

      const timeoutTimer = setTimeout(() => {
        if (isCompleted) return;
        isCompleted = true;
        cleanupJsonp();
        setSyncUiState("error", "LIVE | Lỗi kết nối Script");
        if (!silent) {
          showToast("Không thể kết nối: Script chưa cập nhật Code v3 hoặc chưa chọn quyền 'Bất kỳ ai (Anyone)'.", "error");
        }
      }, 10000);

      function cleanupJsonp() {
        clearTimeout(timeoutTimer);
        if (scriptTag.parentNode) scriptTag.parentNode.removeChild(scriptTag);
        delete window[callbackName];
      }

      window[callbackName] = function (response) {
        if (isCompleted) return;
        isCompleted = true;
        cleanupJsonp();

        if (response && response.status === "ok" && Array.isArray(response.orders)) {
          mergeSheetOrders(response.orders);
          const nowTime = new Date().toLocaleTimeString("vi-VN", { hour: "2-digit", minute: "2-digit" });
          setSyncUiState("live", `LIVE | Đã đồng bộ (${nowTime})`);
          if (!silent) {
            showToast(`Đồng bộ thành công ${response.orders.length} đơn từ Google Sheets!`, "success");
          }
        } else {
          setSyncUiState("error", "LIVE | Lỗi dữ liệu Sheet");
          if (!silent) {
            showToast("Script phản hồi lỗi. Hãy kiểm tra lại tên Sheet trong Code v3.", "error");
          }
        }
      };

      const separator = scriptUrl.includes("?") ? "&" : "?";
      scriptTag.src = `${scriptUrl}${separator}action=getOrders&callback=${callbackName}`;
      scriptTag.onerror = function () {
        if (isCompleted) return;
        isCompleted = true;
        cleanupJsonp();
        setSyncUiState("error", "LIVE | Lỗi kết nối");
        if (!silent) {
          showToast("Không thể kết nối: Script chưa cập nhật Code v3 hoặc chưa chọn quyền 'Bất kỳ ai (Anyone)'.", "error");
        }
      };
      document.body.appendChild(scriptTag);
    }

    function mergeSheetOrders(sheetRows) {
      sheetRows.forEach((row) => {
        const sheetOrderId = `GS-${row.rowIndex}`;
        const existingIdx = orders.findIndex((o) => o.id === sheetOrderId || o.rowIndex === row.rowIndex);
        const unitPrice = row.tier === "student" ? pricingConfig.student : pricingConfig.uni;
        const urgText = (row.urgency || "").toLowerCase();
        let extra = 0;
        let urgencyKey = "normal";
        if (urgText.includes("siêu tốc")) {
          extra = pricingConfig.express;
          urgencyKey = "express";
        } else if (urgText.includes("gấp")) {
          extra = pricingConfig.urgent;
          urgencyKey = "urgent";
        }

        const calcTotal = (row.slideCount || 10) * unitPrice + extra;
        const parsedDeadline = row.deadline && !isNaN(new Date(row.deadline).getTime())
          ? row.deadline
          : new Date(Date.now() + 24 * 3600 * 1000).toISOString();

        if (existingIdx !== -1) {
          orders[existingIdx].customerName = row.customerName || orders[existingIdx].customerName;
          orders[existingIdx].contact = row.contact || orders[existingIdx].contact;
          orders[existingIdx].topic = row.topic || orders[existingIdx].topic;
          orders[existingIdx].status = row.status || orders[existingIdx].status;
          orders[existingIdx].rowIndex = row.rowIndex;
        } else {
          orders.unshift({
            id: sheetOrderId,
            customerName: row.customerName,
            contact: row.contact,
            topic: row.topic,
            tier: row.tier,
            slideCount: row.slideCount,
            urgency: urgencyKey,
            deadline: parsedDeadline,
            totalAmount: calcTotal,
            paidAmount: 0,
            status: row.status || "Mới tiếp nhận",
            note: row.note || "",
            rowIndex: row.rowIndex
          });
        }
      });

      saveOrdersToStorage();
      renderOrders();
      populateQuoteOrderDropdown();
    }

    function formatVND(amount) {
      const num = Number(amount) || 0;
      return num.toLocaleString("vi-VN") + "đ";
    }

    function formatShortDateTime(isoStr) {
      const d = new Date(isoStr);
      if (isNaN(d.getTime())) return "Chưa hẹn";
      const hh = String(d.getHours()).padStart(2, "0");
      const min = String(d.getMinutes()).padStart(2, "0");
      const dd = String(d.getDate()).padStart(2, "0");
      const mm = String(d.getMonth() + 1).padStart(2, "0");
      return `${hh}:${min} ${dd}/${mm}`;
    }

    function toLocalDatetimeInputValue(isoStr) {
      const d = new Date(isoStr);
      if (isNaN(d.getTime())) return "";
      const offsetMs = d.getTimezoneOffset() * 60000;
      return new Date(d.getTime() - offsetMs).toISOString().slice(0, 16);
    }

    function removeVietnameseTones(str) {
      return (str || "")
        .normalize("NFD")
        .replace(/[\u0300-\u036f]/g, "")
        .replace(/đ/g, "d")
        .replace(/Đ/g, "D");
    }

    function escapeHtml(str) {
      return String(str || "")
        .replace(/&/g, "&")
        .replace(//g, ">")
        .replace(/"/g, """)
        .replace(/'/g, "'");
    }
